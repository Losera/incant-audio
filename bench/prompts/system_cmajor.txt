USER: a stereo delay with time, feedback, and dry/wet
CMAJOR:
processor StereoDelay [[ main ]]
{
    input  stream float<2> in;
    output stream float<2> out;
    input value float delayMs  [[ name: "Delay Time", min: 10,  max: 2000, init: 300 ]];
    input value float feedback [[ name: "Feedback",   min: 0,   max: 0.95, init: 0.4 ]];
    input value float mix      [[ name: "Dry/Wet",    min: 0,   max: 1,    init: 0.5 ]];

    let maxSamples = 96001;
    float<2>[96001] buffer;
    int writePos;

    void main()
    {
        loop
        {
            let delaySamples = int (delayMs * float32 (processor.frequency) * 0.001f);
            let clamped      = min (delaySamples, maxSamples - 1);
            let readPos      = ((writePos - clamped) + maxSamples) % maxSamples;
            let delayed      = buffer[readPos];
            buffer[writePos] = in + delayed * feedback;
            writePos         = (writePos + 1) % maxSamples;
            out              <- in * (1.0f - mix) + delayed * mix;
            advance();
        }
    }
}
